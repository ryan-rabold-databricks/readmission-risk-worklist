       count_if(audit_updated_at > TIMESTAMP'2026-11-09T02:05:53Z' AND (first_name IN ('***', 'REDACTED') OR last_name IN ('***', 'REDACTED') OR mrn = '***' OR phone = '***' OR zip_code = '***')) masked_in_those,
